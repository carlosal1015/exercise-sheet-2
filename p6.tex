\section{Pregunta N$^{\circ}$6\qquad Aldo Luna Bueno}

\begin{frame}
	\begin{enumerate}\setcounter{enumi}{5}
		\item

		      Muestre que la sucesión

		      \begin{equation*}
			      I_{0}=\ln\left(\frac{6}{5}\right),\quad
			      \forall k\in\mathbb{N}:
			      I_{k}=\frac{1}{k}-5I_{k-1}
		      \end{equation*}
		      no es apropiada para aproximar la integral
		      \begin{equation*}
			      I_{k}=
			      \int\limits_{0}^{1}
			      \frac{x^{k}}{5+x}
			      \dl x
		      \end{equation*}
		      a pesar que funciona en aritmética infinita.
	\end{enumerate}
	\begin{solution}
		La fórmula es correcta porque
		\begin{math}
			\displaystyle
			I_{\alert{0}}=
			\int\limits_{0}^{1}
			\frac{x^{\alert{0}}}{5+x}
			\dl x=
			\int\limits_{0}^{1}
			\frac{1}{5+x}
			\dl x=
			\int\limits_{5}^{6}
			\frac{1}{u}
			\dl u
			=
			{
			\ln(u)
			\Bigr|
			}_{5}^{6}
			=
			\ln\left(6\right)-
			\ln\left(5\right)=
			\ln\left(\frac{6}{5}\right)
		\end{math}.
		Además,
		\begin{align*}
			I_{\alert{k}} & =
			\int\limits_{0}^{1}
			\frac{x^{\alert{k}}}{5+x}
			\dl x=
			\int\limits_{0}^{1}
			x^{k-1}
			\left(\alert{\frac{x}{5+x}}\right)
			\dl x=
			\int\limits_{0}^{1}
			x^{k-1}
			\left(\alert{1-\frac{5}{5+x}}\right)
			\dl x=
			\int\limits_{0}^{1}
			\left(x^{k-1}-5\frac{x^{k-1}}{5+x}\right)
			\dl x             \\
			I_{\alert{k}} & =
			\int\limits_{0}^{1}
			x^{k-1}
			\dl x-
			5
			\int\limits_{0}^{1}
			\frac{x^{\alert{k-1}}}{5+x}
			\dl x=
			{
			\frac{x^{k-1+1}}{k-1+1}
			\Bigr|
			}_{0}^{1}
			-5I_{\alert{k-1}}=
			\left(1-0\right)-5I_{k-1}.
		\end{align*}
	\end{solution}
\end{frame}
